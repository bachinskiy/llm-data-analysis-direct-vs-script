function transform(rows) {
  function parseDamage(value) {
    if (typeof value !== "string") return null;
    if (!/^(?:\d+|[1-9]\d{0,2}(?:,\d{3})+)$/.test(value)) return null;
    return Number(value.replace(/,/g, ""));
  }

  function parseCount(value) {
    if (typeof value !== "string" || !/^\d+$/.test(value)) return null;
    return Number(value);
  }

  var records = [];
  var severity_counts = { unknown: 0, critical: 0, high: 0, medium: 0, low: 0 };

  for (var i = 0; i < rows.length; i += 1) {
    var row = rows[i];
    var damage = parseDamage(row.damage_text);
    var injured = parseCount(row.injured_text);
    var hazmat = parseCount(row.hazmat_released_text);
    var severity;

    if (damage === null || injured === null || hazmat === null) {
      severity = "unknown";
    } else if (injured > 0 || hazmat > 0 || damage >= 1000000) {
      severity = "critical";
    } else if (damage >= 100000) {
      severity = "high";
    } else if (damage >= 10000) {
      severity = "medium";
    } else {
      severity = "low";
    }

    severity_counts[severity] += 1;
    records.push({
      raw_id: row.raw_id,
      report_year: row.report_year,
      track_type: row.track_type,
      severity_class: severity,
      valid_damage_usd: damage,
      valid_injured: injured,
      valid_hazmat: hazmat
    });
  }

  return {
    records: records,
    summary: {
      total_records: records.length,
      severity_counts: severity_counts
    }
  };
}

module.exports = { transform };
